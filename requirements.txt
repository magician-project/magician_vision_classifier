# Core: training, inference and analysis
# numpy 2.x: must match what torch 2.8.0+cu128 was built against — a 1.x downgrade
# breaks the compiled extensions with "numpy.dtype size changed".
numpy>=2,<3
opencv-python
pillow
# Pinned: the host driver on supermicro-nvidia is 535.183.01 (CUDA 12.3). Unpinned
# torch now resolves to a cu130 (CUDA 13) wheel, which needs driver >=580 and reports
# cuda.is_available()==False. 2.8.0+cu128 is also the exact stack that produced the
# R0 miss@FA5=8.4 result, so this keeps reproductions comparable.
torch==2.8.0
torchvision==0.23.0
# timm: the small/edge backbone families (convnext_atto..nano, repvgg, mobileone,
# ghostnet, lcnet, mobilenetv4, fastvit, edgenext) and timm.utils.reparameterize_model
# for the repvgg/mobileone deployment fusion. Unlike the torchvision branches it also
# ADAPTS the pretrained stem to in_chans instead of discarding it (ModelZoo._stem_like
# reproduces that for the torchvision side).
# PINNED for the same reason as torch: `pretrained=True` resolves to whatever weights
# THIS timm version points at, and its model definitions change between releases, so an
# unpinned timm can silently change what a backbone even is. 1.0.28 produced the whole
# 2026-08 backbone sweep (convnext_pico 7.51 screen / 4.37 full train).
timm==1.0.28
# Pinned: Lightning owns the checkpoint format score_checkpoints.py reads and the
# Trainer semantics (limit_train_batches, save_top_k) the experiment configs rely on.
pytorch-lightning==2.5.5
torchmetrics==1.8.2
h5py
tqdm
psutil

# Metrics and plotting
scikit-learn
matplotlib
seaborn
